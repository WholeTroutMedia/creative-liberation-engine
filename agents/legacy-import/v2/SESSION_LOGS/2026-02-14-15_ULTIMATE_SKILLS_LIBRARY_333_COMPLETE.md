## 🚀 Usage Example

```typescript
import { 
  initializeSkillsSystem,
  SkillPlayground,
  PerformanceOptimizer 
} from './agents/skills-library/integration';

// Initialize (loads all 333 skills)
const { library, executor, analytics } = await initializeSkillsSystem();

console.log(`Loaded ${library.getAllSkills().length} skills`);
// Output: Loaded 333 skills

// Use any skill
const skill = library.getSkill('research-paper-reading');
const result = await executor.execute({
  skill,
  input: { paper: 'attention-is-all-you-need.pdf' },
  agent: 'IRIS'
});

// Interactive playground
const playground = new SkillPlayground(library, executor, analytics);
const session = playground.startSession('AI Research Experiment');

await playground.executeInteractive(
  session.id,
  'research-diffusion-models',
  { dataset: 'imagenet' }
);

// Compare skills
const comparison = await playground.compareSkills(
  session.id,
  ['research-gans', 'research-diffusion-models', 'research-vaes'],
  { task: 'image-generation' }
);

console.log(`Best skill: ${comparison.winner}`);

// Performance optimization
const optimizer = new PerformanceOptimizer(analytics);
const metrics = optimizer.getMetrics();

console.log(`Avg execution: ${metrics.avgExecutionTime}ms`);
console.log(`Cache hit rate: ${metrics.cacheHitRate}%`);
console.log(`Throughput: ${metrics.throughput} exec/s`);

// Get recommendations
const recs = optimizer.generateRecommendations();
recs.forEach(rec => {
  console.log(`[${rec.priority}] ${rec.description}`);
  console.log(`  Impact: ${rec.expectedImpact}`);
});

// Generate report
console.log(playground.generateReport(session.id));
```
