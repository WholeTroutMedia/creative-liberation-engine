### **3. Performance Optimization**
```typescript
// Check cache before execution
const cached = optimizer.checkCache(skillId, input);
if (cached) return cached;

// Execute and cache
const result = await executor.execute(context);
optimizer.cacheResult(skillId, input, result);
optimizer.recordExecutionTime(result.duration);

// Get metrics
const metrics = optimizer.getMetrics();
console.log(`Avg time: ${metrics.avgExecutionTime}ms`);
console.log(`Cache hit rate: ${metrics.cacheHitRate}%`);
console.log(`Throughput: ${metrics.throughput} exec/s`);

// Get recommendations
const recommendations = optimizer.generateRecommendations();
recommendations.forEach(rec => {
  console.log(`[${rec.priority}] ${rec.description}`);
});

// Identify slow skills
const slowSkills = optimizer.identifySlowSkills();
console.log('Slow skills:', slowSkills);
```
